const express = require('express'); const path = require('path'); const sqlite3 = require('sqlite3').verbose(); const { open } = require('sqlite'); const app = express(); const PORT = process.env.PORT || 3000; const dbFile = path.join(__dirname, 'database.sqlite'); const allowedRoles = ['director', 'assistant']; const entityDefinitions = { courses: { name: 'Dersler', fields: [ { key: 'name', label: 'Ders Adı', type: 'text', required: true }, { key: 'code', label: 'Ders Kodu', type: 'text' }, { key: 'teacher', label: 'Öğretmen', type: 'text' }, { key: 'day', label: 'Gün', type: 'text' }, { key: 'hour', label: 'Saat', type: 'text' } ] }, teachers: { name: 'Öğretmenler', fields: [ { key: 'name', label: 'Ad Soyad', type: 'text', required: true }, { key: 'branch', label: 'Branş', type: 'text' }, { key: 'phone', label: 'Telefon', type: 'text' }, { key: 'email', label: 'E-posta', type: 'email' }, { key: 'status', label: 'Durum', type: 'text' } ] }, classes: { name: 'Sınıflar', fields: [ { key: 'name', label: 'Sınıf Adı', type: 'text', required: true }, { key: 'level', label: 'Seviye', type: 'text' }, { key: 'class_teacher', label: 'Sınıf Öğretmeni', type: 'text' }, { key: 'student_count', label: 'Öğrenci Sayısı', type: 'number' }, { key: 'room', label: 'Sınıf No', type: 'text' } ] } }; let db; async function initDatabase() { db = await open({ filename: dbFile, driver: sqlite3.Database }); await db.exec(` CREATE TABLE IF NOT EXISTS courses ( id INTEGER PRIMARY KEY AUTOINCREMENT, role TEXT NOT NULL CHECK (role IN ('director', 'assistant')), name TEXT NOT NULL, code TEXT, teacher TEXT, day TEXT, hour TEXT, created_at TEXT DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE IF NOT EXISTS teachers ( id INTEGER PRIMARY KEY AUTOINCREMENT, role TEXT NOT NULL CHECK (role IN ('director', 'assistant')), name TEXT NOT NULL, branch TEXT, phone TEXT, email TEXT, status TEXT DEFAULT 'Aktif', created_at TEXT DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE IF NOT EXISTS classes ( id INTEGER PRIMARY KEY AUTOINCREMENT, role TEXT NOT NULL CHECK (role IN ('director', 'assistant')), name TEXT NOT NULL, level TEXT, class_teacher TEXT, student_count INTEGER DEFAULT 0, room TEXT, created_at TEXT DEFAULT CURRENT_TIMESTAMP ); `); } app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); function validateRole(role) { return allowedRoles.includes(role); } function validateEntity(entity) { return Object.prototype.hasOwnProperty.call(entityDefinitions, entity); } function validateFields(entity, data) { const fields = entityDefinitions[entity].fields; const cleaned = {}; for (const field of fields) { const value = data[field.key]; if (field.required && (value === undefined || value === null || String(value).trim() === '')) { throw new Error(`${field.label} alanı zorunludur.`); } if (value !== undefined && value !== null && value !== '') { cleaned[field.key] = field.type === 'number' ? Number(value) : String(value).trim(); } } return cleaned; } async function getTableName(entity) { return entity === 'courses' || entity === 'teachers' || entity === 'classes' ? entity : null; } app.get('/api/config', (req, res) => { res.json({ roles: [ { value: 'director', label: 'Müdür' }, { value: 'assistant', label: 'Müdür Yardımcısı' } ], entities: Object.keys(entityDefinitions).map((key) => ({ value: key, label: entityDefinitions[key].name })) }); }); app.get('/api/:role/:entity', async (req, res) => { const { role, entity } = req.params; if (!validateRole(role)) { return res.status(400).json({ message: 'Geçersiz rol.' }); } if (!validateEntity(entity)) { return res.status(400).json({ message: 'Geçersiz tablo.' }); } const table = await getTableName(entity); const rows = await db.all(`SELECT * FROM ${table} WHERE role = ? ORDER BY id DESC`, [role]); res.json(rows); }); app.post('/api/:role/:entity', async (req, res) => { const { role, entity } = req.params; if (!validateRole(role)) { return res.status(400).json({ message: 'Geçersiz rol.' }); } if (!validateEntity(entity)) { return res.status(400).json({ message: 'Geçersiz tablo.' }); } try { const payload = validateFields(entity, req.body); const table = await getTableName(entity); const columns = Object.keys(payload); const placeholders = columns.map(() => '?').join(', '); const values = columns.map((key) => payload[key]); if (!columns.length) { return res.status(400).json({ message: 'En az bir alan doldurulmalıdır.' }); } const result = await db.run( `INSERT INTO ${table} (role, ${columns.join(', ')}) VALUES (?, ${placeholders})`, [role, ...values] ); const inserted = await db.get(`SELECT * FROM ${table} WHERE id = ?`, [result.lastID]); res.status(201).json(inserted); } catch (error) { res.status(400).json({ message: error.message || 'Kayıt eklenirken hata oluştu.' }); } }); app.put('/api/:role/:entity/:id', async (req, res) => { const { role, entity, id } = req.params; if (!validateRole(role)) { return res.status(400).json({ message: 'Geçersiz rol.' }); } if (!validateEntity(entity)) { return res.status(400).json({ message: 'Geçersiz tablo.' }); } try { const payload = validateFields(entity, req.body); const table = await getTableName(entity); const updates = Object.keys(payload).map((key) => `${key} = ?`).join(', '); const values = Object.keys(payload).map((key) => payload[key]); if (!updates) { return res.status(400).json({ message: 'Güncellenecek alan bulunamadı.' }); } const existing = await db.get(`SELECT * FROM ${table} WHERE id = ? AND role = ?`, [id, role]); if (!existing) { return res.status(404).json({ message: 'Kayıt bulunamadı.' }); } await db.run(`UPDATE ${table} SET ${updates} WHERE id = ? AND role = ?`, [...values, id, role]); const updated = await db.get(`SELECT * FROM ${table} WHERE id = ?`, [id]); res.json(updated); } catch (error) { res.status(400).json({ message: error.message || 'Kayıt güncellenirken hata oluştu.' }); } }); app.delete('/api/:role/:entity/:id', async (req, res) => { const { role, entity, id } = req.params; if (!validateRole(role)) { return res.status(400).json({ message: 'Geçersiz rol.' }); } if (!validateEntity(entity)) { return res.status(400).json({ message: 'Geçersiz tablo.' }); } const table = await getTableName(entity); const existing = await db.get(`SELECT * FROM ${table} WHERE id = ? AND role = ?`, [id, role]); if (!existing) { return res.status(404).json({ message: 'Kayıt bulunamadı.' }); } await db.run(`DELETE FROM ${table} WHERE id = ? AND role = ?`, [id, role]); res.json({ message: 'Kayıt başarıyla silindi.' }); }); app.get('*', (req, res) => { res.sendFile(path.join(__dirname, 'public', 'index.html')); }); async function startServer() { try { await initDatabase(); app.listen(PORT, () => { console.log(`Sunucu çalışıyor: http://localhost:${PORT}`); }); } catch (error) { console.error('Sunucu başlatma hatası:', error); process.exit(1); } } startServer();
